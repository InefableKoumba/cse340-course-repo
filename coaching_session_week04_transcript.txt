Here is the processEditProjectForm controller function from src/controllers/projects.js:

const processEditProjectForm = async (req, res) => {
    const projectId = req.params.id;
    const { title, description, location, date, organization_id } = req.body;

    await updateProject(projectId, {
        title,
        description,
        location,
        date,
        organization_id
    });

    res.redirect(`/project/${projectId}`);
};

How it works:
- First, we enable express.urlencoded({ extended: true }) in server.js so Express can parse POST data from HTML forms into req.body.
- In this controller, we get the target ID from req.params.id and extract the submitted form fields from req.body using destructuring.
- We then pass those values into our updateProject model function to safely run the SQL query.
- After the database update finishes, we use res.redirect() to take the user back to the updated details page.

app.use(session({

    saveUninitialized: false,
    cookie: { 
        maxAge: 1000 * 60 * 60 * 24, // 24 hours

app.use(flash());

// Pass flash messages to all EJS views via res.locals
app.use((req, res, next) => {
    res.locals.NODE_ENV = NODE_ENV;
    res.locals.success_msg = req.flash('success');
    res.locals.error_msg = req.flash('error');
    next();
});

// In src/controllers/categories.js: Flashing a message before redirecting
await updateCategory(categoryId, { name: trimmedName, description: trimmedDescription });
req.flash('success', 'Category updated successfully!');
res.redirect(`/category/${categoryId}`);

How it works:
- HTTP is stateless, meaning each request has no memory of past requests. Sessions solve this by storing data on the server linked to a session cookie in the user's browser.
- connect-flash uses the session to store one-time feedback messages across redirects.
- In server.js, we place flash messages into res.locals.success_msg so header.ejs can render the alert box.
- Once rendered, connect-flash automatically removes the message from the session so it doesn't appear again on refresh.

// Validation rules definition
const projectValidation = [
    body('title')
        .trim()
        .notEmpty().withMessage('Title is required')
        .isLength({ min: 3, max: 200 }).withMessage('Title must be between 3 and 200 characters'),
    body('description')
        .trim()
        .notEmpty().withMessage('Description is required')
        .isLength({ max: 1000 }).withMessage('Description must be less than 1000 characters'),
    body('location')
        .trim()
        .notEmpty().withMessage('Location is required')
        .isLength({ max: 200 }).withMessage('Location must be less than 200 characters'),
    body('date')
        .notEmpty().withMessage('Date is required')
        .isISO8601().withMessage('Date must be a valid date format'),
    body('organizationId')
        .notEmpty().withMessage('Organization is required')
        .isInt().withMessage('Organization must be a valid integer')
];

// Checking validation results in processNewProjectForm controller
const errors = validationResult(req);
if (!errors.isEmpty()) {
    errors.array().forEach((error) => {
        req.flash('error', error.msg);
    });
    return res.redirect('/new-project');
}

Why it's necessary and how it works:
- Client-side validation improves user experience, but it can be easily bypassed by disabling JavaScript or using tools like Postman or curl.
- Server-side validation is our true security boundary.
- We sanitize inputs using trim() and enforce data integrity with notEmpty(), min/max length checks, date format checks (isISO8601), and integer checks (isInt).
- If validation fails, we capture all error messages with validationResult(req), pass them into flash messages, and redirect back to the form so the user sees clear feedback.
